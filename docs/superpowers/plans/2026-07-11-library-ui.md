void WLibraryTableView::applyFont(const QFont& font) {
    setFont(font);
    QFontMetrics metrics(font);
    verticalHeader()->setMinimumSectionSize(metrics.height());
    // Resize the 'Played' checkbox and the BPM lock icon.
    // Note: this works well for library font sizes up to ~200% of the original
    // system font's size (that set with Qt5 Settings respectively). Above that,
    // the indicators start to overlap the item text because the painter rectangle

void WLibraryTableView::setTrackTableRowHeight(int rowHeight) {
    m_manualRowHeight = rowHeight;
    if (m_visibleRows > 0) {
        return;
    }
    verticalHeader()->setDefaultSectionSize(math_max(
            rowHeight, verticalHeader()->minimumSectionSize()));
}

void WLibraryTableView::setTrackTableVisibleRows(int visibleRows) {
    m_visibleRows = visibleRows;
    if (m_visibleRows > 0) {
        applyVisibleRows();
    } else {
        applyFont(m_baseTrackTableFont);
        verticalHeader()->setDefaultSectionSize(math_max(
                m_manualRowHeight, verticalHeader()->minimumSectionSize()));
    }
}

void WLibraryTableView::applyVisibleRows() {
    if (m_visibleRows <= 0) {
        return;
    }
    const int viewportHeight = viewport()->height();
    if (viewportHeight <= 0) {
        // Not laid out yet; resizeEvent() will retry.
        return;
    }
    const int rowHeight = math_max(8, viewportHeight / m_visibleRows);
    QFont font = m_baseTrackTableFont;
    font.setPixelSize(math_max(8, static_cast<int>(rowHeight * 0.45)));
    applyFont(font);
    verticalHeader()->setDefaultSectionSize(rowHeight);
}

void WLibraryTableView::resizeEvent(QResizeEvent* event) {
    QTableView::resizeEvent(event);
    applyVisibleRows();
}

- [ ] **Step 3: Add the tracker** in the anonymous namespace (or a new one) near the top of basetracktablemodel.cpp:

```cpp
namespace {

// XDJ traffic-light: tracks the "master" deck for key-compatibility
// highlighting. Master = the currently playing deck that started playing
// most recently (i.e. has been playing for the least wall-clock time).
// No deck playing -> no master -> no highlighting.
class XdjMasterKeyTracker {
  public:
    static XdjMasterKeyTracker& instance() {
        static XdjMasterKeyTracker s_instance;
        return s_instance;
    }

    void registerModel(QObject* pModel, std::function<void()> refresh) {
        ensureInitialized();
        m_models.append(qMakePair(pModel, std::move(refresh)));
    }

    void unregisterModel(QObject* pModel) {
        for (int i = m_models.size() - 1; i >= 0; --i) {
            if (m_models.at(i).first == pModel) {
                m_models.removeAt(i);
            }
        }
    }

    bool isCompatibleKeyText(const QString& keyText) const {
        if (m_compatibleKeys.isEmpty() || keyText.isEmpty()) {
            return false;
        }
        const mixxx::track::io::key::ChromaticKey key =
                KeyUtils::guessKeyFromText(keyText);
        if (key == mixxx::track::io::key::INVALID) {
            return false;
        }
        return m_compatibleKeys.contains(static_cast<int>(key));
    }

  private:
    static constexpr int kMaxDecks = 4;

    struct DeckState {
        std::unique_ptr<ControlProxy> pPlay;
        QString group;
        qint64 playStartMs = -1;
        bool playing = false;
    };

    XdjMasterKeyTracker() = default;

    void ensureInitialized() {
        if (m_initialized) {
            return;
        }
        m_initialized = true;
        m_pContext = std::make_unique<QObject>();
        for (int i = 0; i < kMaxDecks; ++i) {
            const QString group = QStringLiteral("[Channel%1]").arg(i + 1);
            const ConfigKey playKey(group, QStringLiteral("play"));
            if (!ControlObject::exists(playKey)) {
                continue;
            }
            auto& deck = m_decks[i];
            deck.group = group;
            deck.pPlay = std::make_unique<ControlProxy>(playKey, m_pContext.get());
            deck.pPlay->connectValueChanged(m_pContext.get(), [this, i](double v) {
                onPlayChanged(i, v > 0);
            });
            deck.playing = deck.pPlay->toBool();
            if (deck.playing) {
                deck.playStartMs = QDateTime::currentMSecsSinceEpoch();
            }
        }
        QObject::connect(&PlayerInfo::instance(),
                &PlayerInfo::trackChanged,
                m_pContext.get(),
                [this](const QString& group, TrackPointer, TrackPointer) {
                    if (group == m_masterGroup) {
                        updateMaster();
                    }
                });
        updateMaster();
    }

    void onPlayChanged(int deckIndex, bool playing) {
        auto& deck = m_decks[deckIndex];
        if (playing && !deck.playing) {
            deck.playStartMs = QDateTime::currentMSecsSinceEpoch();
        }
        deck.playing = playing;
        updateMaster();
    }

    void updateMaster() {
        // Master = playing deck with the latest start time.
        QString masterGroup;
        qint64 latestStart = -1;
        for (const auto& deck : m_decks) {
            if (deck.playing && deck.playStartMs > latestStart) {
                latestStart = deck.playStartMs;
                masterGroup = deck.group;
            }
        }
        m_masterGroup = masterGroup;

        QSet<int> compatible;
        if (!masterGroup.isEmpty()) {
            const TrackPointer pTrack =
                    PlayerInfo::instance().getTrackInfo(masterGroup);
            if (pTrack) {
                mixxx::track::io::key::ChromaticKey key = pTrack->getKey();
                if (key == mixxx::track::io::key::INVALID) {
                    key = KeyUtils::guessKeyFromText(pTrack->getKeyText());
                }
                if (key != mixxx::track::io::key::INVALID) {
                    const auto keys = KeyUtils::getCompatibleKeys(key);
                    for (const auto compatibleKey : keys) {
                        compatible.insert(static_cast<int>(compatibleKey));
                    }
                }
            }
        }
        if (compatible == m_compatibleKeys) {
            return;
        }
        m_compatibleKeys = compatible;
        for (const auto& entry : std::as_const(m_models)) {
            entry.second();
        }
    }

    bool m_initialized = false;
    std::unique_ptr<QObject> m_pContext;
    DeckState m_decks[kMaxDecks];
    QString m_masterGroup;
    QSet<int> m_compatibleKeys;
    QList<QPair<QObject*, std::function<void()>>> m_models;
};

} // namespace
```
